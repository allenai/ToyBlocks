================================================================================
  ToyBlocks + TAMP — Flip + Pick-and-Place End-to-End Commands
================================================================================

Runs a pick-and-place TAMP task entirely through toyblocks's scripts, backed by
the TAMP planner (tmp3). Validated on the FR3 (franka_droid) robot.



--------------------------------------------------------------------------------
Step 1: Generate the PDDL problem (flip RL + PnP)
--------------------------------------------------------------------------------
# ToyBlocks has NO PDDL-problem generator (its scripts only CONSUME PDDLs).
# Start block_r at o17; goal o2 + on block_red1 -> REORIENT then PNP_TABLE_BLOCK.

cat > /tmp/flip_pnp_r_on_red.pddl <<'EOF'
(define (problem flip-pnp-r-on-red)
    (:domain blocks-tamp)
    (:objects
        block_r - block_r_e
        block_red1 - block
        workspace_table - table
        robot - gripper
    )
    (:init
        (on_table block_r workspace_table)
        (on_table block_red1 workspace_table)
        (clear block_r)
        (clear block_red1)
        (handempty robot)
        (gripper_at_init)
        (ori_of block_r o17)
        (ori_of block_red1 o1)
    )
    (:goal (and
;       (ori_of block_r o2)
        (on block_r block_red1)
    ))
    ;block_size_range = [0.03, 0.05]
    ;table_height_range = [0, 0.04]
)
EOF

Alternative

python TAMP/test_domains/Blocks/generate_letter_color_problem.py \
  --count 500 \
  --seed 42 \
  --out-dir "${ROOT}/datasets/unmerged/${EXP}/tasks" \
  --letter-blocks block_r \
  --color-blocks block_red1 \
  --goal-condition "(on block_r block_red1) (ori_of block_r o2)"


--------------------------------------------------------------------------------
Step 2: Codegen the env package + sample reset states
--------------------------------------------------------------------------------

python scripts/generate_task_from_pddl.py \
    --pddl /tmp/flip_pnp_r_on_red.pddl \
    --domain_path src/toyblocks/tamp/blocks \
    --output_dir datasets/basics/test/ \
    --num_reset_states 16 \
    --num_envs 8 \
    --headless

# Quick single-instance smoke test: use --num_reset_states 1 --num_envs 1


--------------------------------------------------------------------------------
Step 3: Generate plans (TAMP planner, CuRobo-only — no Isaac sim)
--------------------------------------------------------------------------------

python scripts/generate_plans.py \
    --num-workers 4 \
    --domain-name Blocks \
    --domain-path src/toyblocks/tamp/blocks \
    --pddl-problem-file /tmp/flip_pnp_r_on_red.pddl \
    --env-path datasets/basics/test/flip_pnp_r_on_red \
    --device cuda:0 \
    --robotiq


--------------------------------------------------------------------------------
Step 4: Execute the plans in Isaac Sim (records LeRobot data + videos)
--------------------------------------------------------------------------------
# --num_envs       : parallel envs (start small; FR3 visual mesh is heavy)
# --max_policies   : how many plan .pkl files to run
# --record_data    : save LeRobot dataset + videos
# --data_dir       : output (success/ + failure/ with per-camera mp4s)

python scripts/execute_plans.py \
    --policy_dir src/toyblocks/tasks/plans/flip_pnp_r_on_red/ \
    --num_envs 8 \
    --domain_name Blocks \
    --domain_path src/toyblocks/tamp/blocks \
    --pddl_problem_file /tmp/flip_pnp_r_on_red.pddl \
    --env_path datasets/basics/test/flip_pnp_r_on_red \
    --robotiq \
    --enable_cameras \
    --record_data \
    --data_dir datasets/pnp_test \
    --max_policies 8 \
    --output datasets/pnp_test/results.json \
    --headless
